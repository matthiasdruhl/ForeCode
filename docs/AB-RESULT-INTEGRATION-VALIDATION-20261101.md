Reproducible offline peer restoration (ignored node_modules only; reads local package cache, never private custody or runtime installation):

```sh
python3 - <<'PY'
import base64, hashlib, io, json, pathlib, tarfile
lock=json.loads(pathlib.Path('package-lock.json').read_text())
for name, entry in lock['packages'].items():
    if not entry.get('peer') or (pathlib.Path(name)/'package.json').exists(): continue
    if entry.get('os') and 'darwin' not in entry['os']: continue
    if entry.get('cpu') and 'arm64' not in entry['cpu']: continue
    algorithm, encoded=entry['integrity'].split('-', 1)
    assert algorithm == 'sha512'
    digest=base64.b64decode(encoded).hex()
    cached=pathlib.Path.home()/'.npm/_cacache/content-v2/sha512'/digest[:2]/digest[2:4]/digest[4:]
    data=cached.read_bytes()
    assert hashlib.sha512(data).hexdigest() == digest
    target=pathlib.Path(name).resolve()
    assert target.is_relative_to(pathlib.Path('node_modules').resolve())
    with tarfile.open(fileobj=io.BytesIO(data),mode='r:gz') as archive:
        for member in archive.getmembers():
            parts=pathlib.PurePosixPath(member.name).parts
            assert parts[0] == 'package' and '..' not in parts
            assert member.isfile() or member.isdir()
            member.name=str(pathlib.PurePosixPath(*parts[1:]))
            if member.name != '.': archive.extract(member,target)
    print(name, entry['version'], 'lockfile SHA-512 verified; scripts not run')
PY
```
